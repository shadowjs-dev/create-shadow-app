# app

A ShadowJS single-page app.

## Getting started

```bash
npm install
npm run dev
```

## Scripts

- `npm run dev` starts the development server with hot reload.
- `npm run build` builds for production.
- `npm run preview` serves the production build locally.
- `npm run typecheck` type-checks with `tsc --noEmit`.
- `npm run lint` checks the code with Biome.
- `npm run format` formats the code with Biome.
- `npm run test` runs the tests once.
- `npm run test:watch` runs the tests and re-runs them on change.

import { createMemoryHistory } from "@shadow-js/router";
import { accessibleMatchers, render, userEvent } from "@shadow-js/testing/dom";
import { expect, it } from "vitest";
import { App } from "./App";

expect.extend(accessibleMatchers);

const at = (path: string) => createMemoryHistory({ initialEntries: [path] });

it("renders the route of the address", async () => {
  const view = render(() => <App history={at("/about")} />);
  expect(await view.findByRole("heading", { name: "About" })).toBeTruthy();
});

it("navigates with a link and shows the not-found page for an unknown address", async () => {
  const view = render(() => <App history={at("/")} />);
  await userEvent.click(await view.findByRole("link", { name: "About" }));
  expect(await view.findByRole("heading", { name: "About" })).toBeTruthy();

  const missing = render(() => <App history={at("/nope")} />);
  expect(await missing.findByRole("heading", { name: "Page not found" })).toBeTruthy();
});

it("every page has one level-1 heading and named links", async () => {
  const view = render(() => <App history={at("/")} />);
  await view.findByRole("heading", { name: "ShadowJS" });
  expect(view.container.querySelectorAll("h1")).toHaveLength(1);
  for (const link of view.getAllByRole("link")) {
    expect(link).toHaveAccessibleName();
  }
  const missing = render(() => <App history={at("/nope")} />);
  await missing.findByRole("heading", { name: "Page not found" });
  expect(missing.container.querySelectorAll("h1")).toHaveLength(1);
});
=== src/App.tsx ===
import { onCleanup } from "@shadow-js/core";
import { Outlet, type RouterOptions, RouterProvider } from "@shadow-js/router";
import { createAppRouter } from "./router";

/** The routed app. Pass a `history` (a memory history, say) to run it without a browser. */
export function App(props: { history?: RouterOptions["history"] }) {
  const router = createAppRouter(props.history);
  onCleanup(() => router.dispose());
  return (
    <RouterProvider router={router}>
      <Outlet />
    </RouterProvider>
  );
}

import { useMemo, useStore } from "@shadow-js/core";
import { For, Show } from "@shadow-js/web";

export function Home() {
  const [count, setCount] = useStore(0);
  const squares = useMemo(() =>
    Array.from({ length: Math.min(count(), 5) }, (_, i) => (i + 1) ** 2),
  );

  return (
    <section>
      <h1 class="text-3xl font-bold">ShadowJS</h1>
      <p>Count: {count()}</p>
      <button
        type="button"
        class="rounded bg-black px-4 py-2 text-white"
        onClick={() => setCount((n) => n + 1)}
      >
        Increment
      </button>
      <Show when={count() > 0} fallback={<p>Press the button.</p>}>
        <ul>
          <For each={squares()}>{(value) => <li>{value}</li>}</For>
        </ul>
      </Show>
    </section>
  );
}

=== src/router.tsx ===
import { createRouter, defineRoute, type RouterOptions } from "@shadow-js/router";
import { About } from "./pages/About";
import { Home } from "./pages/Home";
import { Layout } from "./pages/Layout";
import { NotFound } from "./pages/NotFound";

const home = defineRoute("/", { component: Home });
const about = defineRoute("/about", { component: About });
const missing = defineRoute("*", { component: NotFound });
const root = defineRoute("", {
  id: "root",
  component: Layout,
  children: [home, about, missing],
});

/** Creates the router of the app. Tests pass a memory history. */
export function createAppRouter(history?: RouterOptions["history"]) {
  return createRouter({ routes: [root], ...(history === undefined ? {} : { history }) });
}
