=== README.md ===
# app

A ShadowJS single-page app.

## Getting started

```bash
npm install
npm run dev
npx convex dev   # in a second terminal: log in, create a deployment, write VITE_CONVEX_URL
```

## Scripts

- `npm run dev` starts the development server with hot reload.
- `npm run build` builds for production.
- `npm run preview` serves the production build locally.
- `npm run typecheck` type-checks with `tsc --noEmit`.
- `npm run lint` checks the code with Biome.
- `npm run format` formats the code with Biome.
- `npm run test` runs the tests once.
- `npm run test:watch` runs the tests and re-runs them on change.
- `npm run lint:eslint` checks the code with the ShadowJS ESLint rules.
- `npm run convex` runs `convex dev`.

## UI components

Components are copied into `src/components/ui` so you own them: `npx shadow-ui add button`. The theme (colors, radius, dark mode) is imported in `src/index.css`; `@/` points at `src/`. `cn()` in `src/lib/utils.ts` merges class names.

## ESLint

`npm run lint:eslint` runs the recommended rules of `@shadow-js/eslint-plugin`. The plugin parses TypeScript and JSX itself, so it works with the TypeScript 7 this app compiles with.

## Devtools

In development a launcher button opens the inspector; `Alt+Shift+D` toggles it. It is imported only when `import.meta.env.DEV` is true, so the production build does not contain it.

## Convex

Run `npx convex dev` in a second terminal. The first time it asks you to log in, creates a deployment, writes `VITE_CONVEX_URL` to `.env.local` and keeps `convex/_generated` up to date while you edit the functions in `convex/`.

- `convex/schema.ts` defines the tables, `convex/tasks.ts` holds a query and a mutation.
- `src/pages/Tasks.tsx` lists the tasks with `useQuery` and adds one with `useMutation`.
- `convex/_generated` starts as a stand-in with the same exports, so the app type-checks and builds before the first `convex dev`; Convex replaces it.
- `convex/auth.config.ts` trusts the Clerk JWT template named `convex`: see the Clerk section.

## Clerk

1. Create an application at https://dashboard.clerk.com and copy its keys into `.env.local` (see `.env.example`).
2. `/dashboard` shows its content to signed-in users only (`SignedIn`, `RedirectToSignIn`). In a single-page app the real protection is on the server your app talks to.
3. In the Clerk dashboard create a JWT template named `convex`, then run `npx convex env set CLERK_JWT_ISSUER_DOMAIN <the Issuer of that template>`. `convex/auth.config.ts` trusts it and `ConvexProviderWithAuth` sends the token, so `ctx.auth.getUserIdentity()` works in your functions.

import { v } from "convex/values";
import { mutation, query } from "./_generated/server";

const MAX_TITLE = 200;

export const list = query({
  args: {},
  handler: async (ctx) => {
    return await ctx.db.query("tasks").order("desc").take(50);
  },
});

export const create = mutation({
  args: { title: v.string() },
  handler: async (ctx, args) => {
    const identity = await ctx.auth.getUserIdentity();
    if (identity === null) throw new Error("Sign in to add a task.");
    const title = args.title.trim();
    if (title === "" || title.length > MAX_TITLE) {
      throw new Error("A title has 1 to 200 characters.");
    }
    return await ctx.db.insert("tasks", {
      title,
      done: false,
      ...(identity === null ? {} : { author: identity.subject }),
    });
  },
});

import { Outlet, type RouterOptions, RouterProvider } from "@shadow-js/router";

/** The routed app. Pass a `history` (a memory history, say) to run it without a browser. */
export function App(props: { history?: RouterOptions["history"] }) {
  const router = createAppRouter(props.history);
  onCleanup(() => router.dispose());
  return (
    <RouterProvider router={router}>
      <Outlet />
    </RouterProvider>
  );

import { clsx } from "clsx";
import { twMerge } from "tailwind-merge";

/**
 * Anything a ShadowJS `class` attribute accepts: strings, numbers, nested (readonly) arrays and
 * `{ name: condition }` objects. Falsy values are skipped.
 * @since 1.0.0
 */
export type ClassValue =
  | string
  | number
  | bigint
  | boolean
  | null
  | undefined
  | readonly ClassValue[]
  | { readonly [name: string]: unknown };

/**
 * Joins class names (strings, arrays, `{ name: condition }` objects) and resolves Tailwind
 * conflicts, so the last utility wins: `cn("px-2", props.class)` lets the caller override `px-2`.
 *
 * @param inputs - Any mix of class values; falsy values are skipped.
 * @returns One class string with conflicting Tailwind utilities removed.
 * @example
 * ```ts
 * import { cn } from "@/lib/utils";
 *
 * const active = true;
 * export const classes = cn("px-2 py-1", active && "bg-primary", "px-4");
 * // "py-1 bg-primary px-4"
 * ```
 * @since 1.0.0
 * @see {@link ClassValue}
 */
export function cn(...inputs: ClassValue[]): string {
  return twMerge(clsx(...(inputs as Parameters<typeof clsx>)));
}

import { useMemo, useStore } from "@shadow-js/core";
import { For, Show } from "@shadow-js/web";

export function Home() {
  const [count, setCount] = useStore(0);
  const squares = useMemo(() =>
    Array.from({ length: Math.min(count(), 5) }, (_, i) => (i + 1) ** 2),
  );

  return (
    <section>
      <h1 class="text-3xl font-bold">ShadowJS</h1>
      <p>Count: {count()}</p>
      <button
        type="button"
        class="rounded bg-black px-4 py-2 text-white"
        onClick={() => setCount((n) => n + 1)}
      >
        Increment
      </button>
      <Show when={count() > 0} fallback={<p>Press the button.</p>}>
        <ul>
          <For each={squares()}>{(value) => <li>{value}</li>}</For>
        </ul>
      </Show>
    </section>
  );
}

import { useMutation, useQuery } from "@shadow-js/convex";
import { useStore } from "@shadow-js/core";
import { For, Show } from "@shadow-js/web";
import { api } from "../../convex/_generated/api";

export function TasksPage() {
  const tasks = useQuery(api.tasks.list, {});
  const addTask = useMutation(api.tasks.create);
  const [title, setTitle] = useStore("");

  function submit(event: SubmitEvent) {
    event.preventDefault();
    const value = title().trim();
    if (value === "") return;
    setTitle("");
    void addTask({ title: value });
  }

  return (
    <section>
      <h1 class="text-3xl font-bold">Tasks</h1>
      <form onSubmit={submit}>
        <input
          class="my-2 block w-full rounded border p-2"
          placeholder="New task"
          value={title()}
          onInput={(event) => setTitle(event.currentTarget.value)}
        />
        <button
          type="submit"
          class="rounded bg-black px-4 py-2 text-white"
          disabled={addTask.pending()}
        >
          Add
        </button>
      </form>
      <Show when={tasks()} fallback={<p>Loading</p>}>
        {(list) => (
          <ul class="my-4 list-disc pl-6">
            <For each={list()}>{(task) => <li>{task.title}</li>}</For>
          </ul>
        )}
      </Show>
    </section>
  );
}

import { ClerkProvider, clerkConvexAuth } from "@shadow-js/clerk";
import { ConvexProviderWithAuth, createConvexClient } from "@shadow-js/convex";
import { ErrorBoundary, type ParentProps } from "@shadow-js/web";
import { requireEnv } from "./lib/env";

const convex = createConvexClient(requireEnv("VITE_CONVEX_URL", import.meta.env.VITE_CONVEX_URL));
const publishableKey = requireEnv(
  "VITE_CLERK_PUBLISHABLE_KEY",
  import.meta.env.VITE_CLERK_PUBLISHABLE_KEY,
);
const loadOptions = {
  signInUrl: "/sign-in",
  signUpUrl: "/sign-up",
  signInFallbackRedirectUrl: "/dashboard",
  signUpFallbackRedirectUrl: "/dashboard",
};
// Convex asks Clerk for a token made from the JWT template named "convex".
const useConvexAuth = clerkConvexAuth({ template: "convex" });

function ConvexBridge(props: ParentProps) {
  return (
    <ConvexProviderWithAuth client={convex} useAuth={useConvexAuth}>
      {props.children}
    </ConvexProviderWithAuth>
  );
}

function clerkFailed(error: unknown, reset: () => void) {
  return (
    <div role="alert">
      <p>Clerk could not be loaded: {error instanceof Error ? error.message : String(error)}</p>
      <button type="button" onClick={reset}>
        Try again
      </button>
    </div>
  );
}

export function Providers(props: ParentProps) {
  return (
    <ErrorBoundary fallback={clerkFailed}>
      <ClerkProvider publishableKey={publishableKey} loadOptions={loadOptions}>
        <ConvexBridge>{props.children}</ConvexBridge>
      </ClerkProvider>
    </ErrorBoundary>
  );
}
